



idle :                     if (convolution_t = VERTICAL) then
                                    y = img_offset_up;
                           else
                                    y = 0;    
                           goto convolution_y_loop_start;
                                                                               
convolution_y_loop : if (y >= img_height - img_offset_down && convolution_t = VERTICAL) then
                                goto stop; 
                           else if (y >= img_height - img_offset_down - img_offset_up && convolution_t = HORIZONTAL) then
                                goto stop;
                           else
                                goto convolution_x_loop_start;  

convolution_x_loop : if (x >= img_width) then 
                                y = y + 1;
                                x = 0; 
                                goto convolution_y_loop_start;
                           else   
                                sum1 = 0;
                                sum2 = 0;                            
                                goto convolution_k_loop_start; 

convolution_k_loop:  if (k >= sigma_size) then
                                x = x + 1;
                                k = 0;
                                goto convolution_x_loop_start;
                           else
                                goto convolution_add_and_shift;  
                                        
convolution_sum_calc: sum1 += pix1 * val;
                           sum2 += pix2 * val;       
                           k = k + 1;
                           goto convolution_k_loop_start;

stop:                      nop;
