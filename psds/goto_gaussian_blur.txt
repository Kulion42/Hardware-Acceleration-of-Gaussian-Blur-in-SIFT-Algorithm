if (start == 1 && ready == 1 && reset == 0)
ready = 0;

else if (start == 0 && ready == 0 && reset == 0){
sigma_t sigma = read_rom(VP_ADDR_SIGMA_ROM_L + img_per_octave);
int size = std::ceil(6 * sigma);

check_size:
if (size % 2 == 0)
  size++;

//-------------------------------

int center = size / 2;


sigm_t sum_kernel = 0;

//-------------------------------


//for (int k = -size/2; k <= size/2; k++)

int k = -size / 2;
kernel_init_loop_start:
if (k > size / 2) goto kernel_init_loop_end;
data_t val1 = std::exp(-(k * k) / (2 * sigma * sigma));
data_t val2 = 0;
write_mem(VP_ADDR_KERNEL_BRAM_L + 2 *(center+k), val1, val2);
sum_kernel += val1;
k++;
goto kernel_init_loop_start;

//-------------------------------

kernel_init_loop_end:


//for (int k = 0; k < size; k++)

k = 0;
kernel_normalization_loop_start:
if (k >= size) goto kernel_normalization_loop_end;
data_t val1, val2;
read_mem(VP_ADDR_KERNEL_BRAM_L + 2 * k, val1, val2);
val1/=sum_kernel;
write_mem(VP_ADDR_KERNEL_BRAM_L + 2 * k, val1, val2);
k++;
goto kernel_normalization_loop_start;

//-------------------------------

kernel_normalization_loop_end:

//-------------------------------


//for (int x = 0; x < img.width; x++)

int x = 0;
vertical_convolution_loop_start:
if (x >= img.width) goto vertical_convolution_loop_end;


//for (int y = 0; y < img.height; y++)

int y =  img_offset_up;
vertical_convolution_inner_loop_start:
if (y >= img.height -  img_offset_down) goto vertical_convolution_inner_loop_end;
sum = 0;

sum_t sum1= 0; sum_t sum2 = 0;
//for (int k = 0; k < size; k++)

k = 0;
vertical_convolution_kernel_loop_start:
if (k >= size) goto vertical_convolution_kernel_loop_end;
int c_y;
if (y+dy < 0 && img_offset_up == 0)
   c_y = 0;
else if (y+dy < img_offset_up == 10)
   c_y = img_offset_up + dy;                 
else if (y+dy >= img_height && img_offset_down == 0)
   int c_y = img_height - 1;
else if (y+dy >= img_height-img_offset_down && img_offset_down == 10)
   c_y =  img_height-img_offset_down + dy;
else
   c_y = y + dy;
   
data_t pix1, pix2, val1, val2;

read_mem(VP_ADDR_KERNEL_BRAM_L + 2 * k, val1, val2);

read_mem(VP_ADDR_MAIN_BRAM_L + 2 *(c_y*img_width + x), pix1, pix2);
sum1 += pix1 * val1;
sum2 += pix2 * val1;

k++;
goto vertical_convolution_kernel_loop_start;

//-------------------------------

vertical_convolution_kernel_loop_end:
write_mem(VP_ADDR_TMP_BRAM_L + 2 *((y-img_offset_up)*img_width + x), sum1, sum2);
y++;
goto vertical_convolution_inner_loop_start;

//-------------------------------

vertical_convolution_inner_loop_end:
x+=2;
goto vertical_convolution_loop_start;

//-------------------------------

vertical_convolution_loop_end:

//for (int x = 0; x < img.width; x++)

x = 0;
horizontal_convolution_loop_start:
if (x >= img.width) goto horizontal_convolution_loop_end;


//for (int y = 0; y < img.height; y++)

y = 0;
horizontal_convolution_inner_loop_start:
if (y >= img.height- img_offset_up - img_offset_down) goto horizontal_convolution_inner_loop_end;
sum_t sum1 = 0; sum_t sum2 = 0;


//for (int k = 0; k < size; k++)

k = 0;
horizontal_convolution_kernel_loop_start:
if (k >= size) goto horizontal_convolution_kernel_loop_end;
int dx = -center + k;
int c_x;
if (x+dx < 0 )
  c_x = 0;
else if (x+dx >= img_width)
  c_x = img_width - 1;
else
  c_x = x + dx;
data_t pix1, pix2, val1, val2;
read_mem(VP_ADDR_KERNEL_BRAM_L + 2 * k, val1, val2);

read_mem(VP_ADDR_TMP_BRAM_L + 2 * (y*img_width + c_x), pix1, pix2);
                 
sum1 += pix1 * val1;
sum2 += pix2 * val1;
 
k++;
goto horizontal_convolution_kernel_loop_start;

//-------------------------------

horizontal_convolution_kernel_loop_end:
write_mem(VP_ADDR_MAIN_BRAM_L + 2 * (y*img_width + x), sum1, sum2);
y++;
goto horizontal_convolution_inner_loop_start;

//-------------------------------

horizontal_convolution_inner_loop_end:
x+=2;
goto horizontal_convolution_loop_start;

//-------------------------------


horizontal_convolution_loop_end:

ready = 1;
}

else if (reset==1){
ready = 0;

sc_dt::sc_uint<64> k = VP_ADDR_MAIN_BRAM_L;

main_bram_empty_start:
if (k < VP_ADDR_MAIN_BRAM_H)  goto main_bram_empty_end;      

write_mem(k, data_t(0.0),data_t(0.0));

k+=4;
goto main_bram_empty_start;

main_bram_empty_end;
//-----------------------------------
sc_dt::sc_uint<64> k = VP_ADDR_TMP_BRAM_L;

tmp_bram_empty_start:
if (k < VP_ADDR_MAIN_BRAM_H)  goto tmp_bram_empty_end;      

write_mem(k, data_t(0.0),data_t(0.0));

k+=4;
goto tmp_bram_empty_start;

main_tmp_empty_end;
//---------------------------------------
sc_dt::sc_uint<64> k = VP_ADDR_KERNEL_BRAM_L;

kernel_bram_empty_start:
if (k < VP_ADDR_MAIN_BRAM_H)  goto kernel_bram_empty_end;      

write_mem(k, data_t(0.0),data_t(0.0));

k+=4;
goto kernel_bram_empty_start;

kernel_bram_empty_end;

  
   ready = 1;
   }
